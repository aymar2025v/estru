
// FIBONACI

#include <iostream>

int fibonaci(int n){
    if(n == 0) return 0;
    if(n == 1) return 1;
    return fibonaci(n - 1) + fibonaci(n - 2);
}

int main()
{
    int x;
    std::cout << "ingrese un indice: ";
    std::cin >> x;
    
    for(int i = 0; i < x; i++){
        std::cout<< fibonaci(i) << " ";
    }

    return 0;
}
