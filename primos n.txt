#include <iostream>
using namespace std;

bool esPrimo(int numero, int divisor = 2) {
    
    if (numero <= 1) return false;
    if (divisor * divisor > numero) return true;
    if (numero % divisor == 0) return false;

        return esPrimo(numero, divisor + 1);
}

int main() {
    int n;

    cout << "ingrese la cantidad de primos a generar: ";
    cin >> n;
    
    if (n <= 0) {
        cout << "error: ingrese un numero positivo." << endl;
        return 1;
    }
    
    cout << "Los primeros " << n << " números primos son: ";
    
    int contador = 0;      
    for (int numero = 2; contador < n; numero++) {
        if (esPrimo(numero)) {
            cout << numero;
            contador++;
            
            if (contador < n) {
                cout << " ";
            }
        }
    }
    
    return 0;
}
