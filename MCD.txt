#include <iostream>
using namespace std;

int mcd(int a, int b) {
    if (b == 0)
        return a;
    else
        return mcd(b, a % b);
}

int main() {
    int a, b;

    cout << "Ingrese dos numeros: ";
    cin >> a >> b;

    cout << "El MCD es: " << mcd(a, b) << endl;

    return 0;
}
