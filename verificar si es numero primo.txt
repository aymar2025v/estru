#include <iostream>
using namespace std;

bool esPrimo(int numero, int divisor = 2) {
    if (numero <= 1) {
        return false;
    }

    if (divisor * divisor > numero) {
        return true;
    }

    if (numero % divisor == 0) {
        return false;
    }

    return esPrimo(numero, divisor + 1);
}

int main() {
    
    int n;
    cout << "Ingrese un numero entero positivo: ";
    cin >> n;
    
    if (esPrimo(n)) {
        cout << n << " ES un numero primo." << endl;
    } else {
        cout << n << " NO es un numero primo." << endl;
    }
    
    return 0;
}
